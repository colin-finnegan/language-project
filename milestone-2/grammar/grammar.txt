program = block

block = statement block
      |

statement = print
          | level1

print = "print" level1

// Assignment
level1 = VARIABLE "=" level1
           | level2

// Logical OR
level2 = level2 "||" level3
           | level3

// Logical AND
level3 = level3 "&&" level4
            | level4

// Relational    
level4 = level4 "==" level5
           | level4 "!=" level5
           | level4 "<"  level5
           | level4 ">"  level5
           | level4 "<=" level5
           | level4 ">=" level5
           | level5

// Additive (+ -)
level5 = level5 "+" level6
         | level5 "-" level6
         | level6

// Multiplicative (* / % & | ^ << >>)
level6 = level6 "*" level7
               | level6 "/" level7
               | level6 "%" level7
               | level6 "&" level7
               | level6 "|" level7
               | level6 "^" level7
               | level6 "<<" level7
               | level6 ">>" level7
               | level7

// Power (**)
level7 = level8 "**" level7
      | level8

// Unary
level8 = "!" level8
      | "-" level8
      | "~" level8
      | casting
      | primary

casting = "floattoint" "(" level1 ")"
        | "inttofloat" "(" level1 ")"

primary = INTEGER
        | FLOAT
        | STRING
        | BOOLEAN
        | VARIABLE
        | "(" level1 ")"
