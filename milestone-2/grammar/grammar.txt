program = block

block = statement block
      |

statement = print
          | ifStmt
          | whileStmt
          | forStmt
          | functionDef
          | returnStmt
          | level1



// Updated Milestone 4 constructs in Recursive Style
ifStmt = "if" level1 "then" block elseStmt "end"
elseStmt = "else" block 
            | 

whileStmt = "while" level1 block "end"

forStmt = "for" VARIABLE "in" "[" level1 "," level1 "]" block "end"

functionDef = "def" VARIABLE "(" parameters ")" block "end"
parameters = VARIABLE parameterTail 
            |
parameterTail = "," VARIABLE parameterTail 
            |

functionCall = VARIABLE "(" arguments ")"
arguments = level1 argumentTail 
            |
argumentTail = "," level1 argumentTail 
            |

returnStmt = "return" level1

print = "print" level1

// Assignment
level1 = VARIABLE "=" level1
           | level2

// Logical OR
level2 = level2 "||" level3
           | level3

// Logical AND
level3 = level3 "&&" level4
            | level4

// Relational    
level4 = level4 "==" level5
           | level4 "!=" level5
           | level4 "<"  level5
           | level4 ">"  level5
           | level4 "<=" level5
           | level4 ">=" level5
           | level5

// Additive (+ -)
level5 = level5 "+" level6
         | level5 "-" level6
         | level6

// Multiplicative (* / % & | ^ << >>)
level6 = level6 "*" level7
               | level6 "/" level7
               | level6 "%" level7
               | level6 "&" level7
               | level6 "|" level7
               | level6 "^" level7
               | level6 "<<" level7
               | level6 ">>" level7
               | level7

// Power (**)
level7 = level8 "**" level7
      | level8

// Unary
level8 = "!" level8
      | "-" level8
      | "~" level8
      | casting
      | primary

casting = "floattoint" "(" level1 ")"
        | "inttofloat" "(" level1 ")"

primary = INTEGER
        | FLOAT
        | STRING
        | BOOLEAN
        | VARIABLE
        | functionCall
        | "(" level1 ")"
